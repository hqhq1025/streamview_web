<project_specification>
  <project_name>YouTube Studio Clone</project_name>

  <overview>
    YouTube Studio is a comprehensive creator dashboard for managing YouTube channels, content, analytics, and community engagement. This clone will replicate the core functionality including content management (videos, shorts, live streams, posts), video editing, analytics dashboards, community moderation, subtitle management, monetization tracking, channel customization, audio library access, and comprehensive settings. The application features a responsive design with persistent sidebar navigation, tab-based content organization, modal dialogs for settings, and real-time data updates.
  </overview>

  <technology_stack>
    <frontend>
      <framework>React 18+ with TypeScript</framework>
      <styling>Tailwind CSS 3.4+ with custom component library</styling>
      <state_management>React Context API + React Query for server state</state_management>
      <routing>React Router 6+ with nested routes</routing>
      <charts>Recharts or Chart.js for analytics visualization</charts>
      <rich_text_editors>TipTap or similar for description editing</rich_text_editors>
      <video_handling>HTML5 Video API with custom controls</video_handling>
      <build_tool>Vite 5+ for fast development and optimized builds</build_tool>
      <component_library>Custom Headless UI components following YouTube patterns</component_library>
    </frontend>
    <backend>
      <runtime>Node.js 20+ with Express.js</runtime>
      <database>PostgreSQL for structured data + MongoDB for analytics timeseries</database>
      <authentication>OAuth 2.0 with JWT tokens</authentication>
      <file_storage>AWS S3 or similar for video/thumbnail storage</file_storage>
      <video_processing>FFmpeg for video transcoding and thumbnail generation</video_processing>
      <api_style>RESTful API with OpenAPI 3.0 specification</api_style>
      <realtime_features>WebSocket integration for live chat and realtime analytics</realtime_features>
      <background_jobs>Bull/BullMQ with Redis for job queues</background_jobs>
    </backend>
    <communication>
      <api>REST API with JSON responses</api>
      <websocket>Socket.io for real-time features (comments, analytics, live streaming)</websocket>
      <cdn>CloudFront or Cloudflare for static asset delivery</cdn>
    </communication>
  </technology_stack>

  <prerequisites>
    <environment_setup>
      Node.js 20+ and npm/yarn/pnpm installed
      PostgreSQL 14+ with configured database
      MongoDB 6+ for analytics data
      Redis 7+ for caching and job queues
      FFmpeg installed system-wide for video processing
      AWS S3 bucket or equivalent storage configured
      OAuth 2.0 credentials for YouTube API integration (if importing data)
      Git for version control
      Modern web browser (Chrome, Firefox, Edge, Safari)
      IDE with TypeScript support (VS Code recommended)
    </environment_setup>
